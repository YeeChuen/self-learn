class Person {
    var alive = true
    
    let firstName: String
    let lastName: String
    
    init(firstName: String, lastName: String) { // <-- constructor, similar to python
        self.firstName = firstName
        self.lastName = lastName
    }
}
class twelveYearOld: Person {
    static let age = 12 // <-- every person is now age of 12
    
    func introduce() {
        print("I am " + self.firstName  + " " + self.lastName + ", " + String(twelveYearOld.age) + " years old!")
    }
}

var age = twelveYearOld.age
print("all person is age of: " + String(age)) // <-- you are able to access static variable without invoking the class
var lilDave = twelveYearOld(firstName: "Dave", lastName: "Bonny")
lilDave.introduce()
print(lilDave.firstName)
print(type(of: lilDave))

/*
when you do assignment to an instanc (already declared object using class or struct).
a class is reference type, 
a struct is value type

class support inheritance, struct does not
you can always modify property of a class even if it is let, but not in struct, struct is more strict
class must define a constructor, struct does not
if you want to modify variable in struct using built in function, you need add keyword "mutating"

example:
let classInstance = MyClass()
let structInstance = MyStruct()

let anotherClassInstance = classInstance
let anotherStructInstance = structInstance

anotherClassInstance.value = 10
anotherStructInstance.value = 10

print(classInstance.value) // Output: 10 (shared reference)
print(structInstance.value) // Output: 0 (copied value)
*/

// PROTOCOLS
print("PROTOCOLS\n")
// similar to Java Interfaces, used as blueprint/requirement for other classes
protocol CarProto {
    var color: String {get set} // <-- the class with must have a get and set for color
    func drive() -> Void
    func isDriving() -> Bool
}
class Car: CarProto {
    var color: String // <-- this must have a value, either set a default of use init
    var driving: Bool = false
    init(color: String) {
        self.color = color
    }
    
    func drive() -> Void {
        self.driving = true
    }
    func isDriving() -> Bool {
        return self.driving
    }
    // can add more than protocol defined, but MUST conform to protocol
    func stop() {
        self.driving = false
    }
}
var protonCar: Car = Car(
    color: "silver"
)
print(protonCar.isDriving())
protonCar.drive()
print(protonCar.isDriving())
protonCar.stop()
print(protonCar.isDriving())
